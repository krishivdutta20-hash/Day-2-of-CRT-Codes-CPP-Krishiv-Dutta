#include <iostream>
using namespace std;
int main() {
//takes an int in as an input print all number from 1 to n but skip any no. that is multiple of 3 using continuing statement.
     int n;
     cout<<"Enter value of n :";
     cin>>n;
     
    for(int i = 0; i <= n ;i++){
        if (i % 3 == 0){
            continue;
        }
        else{
            cout<<"  \n"<<i;
        }
    }
    return 0;
}
