#include <iostream>
using namespace std;

int main() {
    //takes a year as an input and checkes if it is a leap year or not 4 ,=!100,400
    int year;
    cout<<"Enter a year : ";
    cin>>year;
    
    if( (year % 4 == 0 && year % 100 != 0)||(year % 400 == 0) ){
        cout<<"Year "<<year<<" is a Leap year!";
    }
    else{
        cout<<"Year "<<year<<" is not a Leap year!";
    }

    

    return 0;
    
}
