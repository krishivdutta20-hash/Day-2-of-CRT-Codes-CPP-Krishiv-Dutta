#include <iostream>
using namespace std;
int main() {
    int n;
    cout<<"Enter your percentage : ";
    cin>>n;
 
    if (90 <= n ){
        cout<<" Your grade is A";
    }
    else if (80 <= n && n <= 89){
        cout<<"Your grade is B";
    }
    else if (70 <= n && n <= 79){
        cout<<"Your grade is C";
    }
    else if (60 <= n && n <= 69){
        cout<<"Your grade is D";
    }
    else if (n < 60){
        cout<<"You are fail!";
    }
    return 0;
}
